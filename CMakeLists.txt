cmake_minimum_required(VERSION 3.16)
project(MyTinyWebServer LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "MyTinyWebServer requires Linux (epoll).")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MYSQL QUIET IMPORTED_TARGET mysqlclient)
if(NOT MYSQL_FOUND)
    pkg_check_modules(MYSQL REQUIRED IMPORTED_TARGET libmariadb)
endif()
add_executable(MyTinyWebServer
    src/main.cpp src/log.cpp src/webserver.cpp src/sql_pool.cpp
    src/epoller.cpp src/httpconnector.cpp src/buffer.cpp
    src/httprequest.cpp src/httpresponse.cpp src/httpdata.cpp)
target_include_directories(MyTinyWebServer PRIVATE include log)
target_compile_options(MyTinyWebServer PRIVATE -Wall -Wextra)
target_compile_definitions(MyTinyWebServer PRIVATE SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(MyTinyWebServer PRIVATE Threads::Threads PkgConfig::MYSQL)
